class string : public std::string {
using std::string::string;
public:
explicit string(const std::string &str) : std::string(str) {}
string operator*(int times) const {
rush::clampl(times, 0);
string r;
r.reserve(string::size() * times);
while(times-- > 0) {
r += *this;
}
return r;
}
